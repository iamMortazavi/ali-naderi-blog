{{- /* Images in posts: page-bundle images become responsive, lazy-loaded WebP with fixed dimensions (no layout shift). Audio and video attachments (![](episode.mp3), ![[clip.mp4]]) become players. PDFs become links. */ -}} {{- $u := urls.Parse .Destination -}} {{- $ext := path.Ext $u.Path | lower -}} {{- $res := "" -}} {{- if not $u.IsAbs -}}{{- $res = .PageInner.Resources.Get $u.Path -}}{{- end -}} {{- $src := .Destination -}} {{- with $res -}}{{- $src = .RelPermalink -}}{{- end -}} {{- if in (slice ".mp3" ".m4a" ".ogg" ".oga" ".wav") $ext -}} {{- else if in (slice ".mp4" ".webm") $ext -}} {{- else if eq $ext ".pdf" -}} {{ .Text | default (path.Base $u.Path) }} {{- else if and $res (not (in (slice "svg" "gif") $res.MediaType.SubType)) -}} {{- $w := $res.Width -}} {{- $small := $res.Resize (printf "%dx webp" (int (math.Min $w 480))) -}} {{- $large := $res.Resize (printf "%dx webp" (int (math.Min $w 960))) -}} {{- else -}} {{- end -}}
